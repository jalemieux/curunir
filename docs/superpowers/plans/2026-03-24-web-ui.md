```python
_MAX_ATTACHMENT_CONTENT_SIZE = 512 * 1024  # 512KB


def _enrich_attachments(attachments: list[dict], project_root: str) -> None:
    """Add content and normalize paths for text-based attachments in-place."""
    for att in attachments:
        path = att["path"]
        mime = att.get("mime_type", "")
        is_text = mime.startswith("text/") or mime == "application/json"

        # Normalize path to relative
        if os.path.isabs(path):
            try:
                att["path"] = os.path.relpath(path, project_root)
            except ValueError:
                pass  # different drive on Windows, keep absolute

        if not is_text:
            att["content"] = None
            continue

        if not os.path.isfile(path):
            att["content"] = None
            att["error"] = "file not found"
            continue

        if os.path.getsize(path) > _MAX_ATTACHMENT_CONTENT_SIZE:
            att["content"] = None
            continue

        try:
            with open(path, "r", encoding="utf-8", errors="replace") as f:
                att["content"] = f.read()
        except OSError:
            att["content"] = None
            att["error"] = "file not found"
```

2. **Connection with exponential backoff:**
```javascript
const BACKOFF_BASE = 1000;
const BACKOFF_MAX = 30000;
let ws = null;
let backoffDelay = BACKOFF_BASE;
let attempt = 0;
let pendingPayload = null;
let ready = true;

function connect() {
    ws = new WebSocket(uri);
    ws.onopen = () => {
        if (attempt > 0) addSystemMessage(`Reconnected to ${uri}`);
        backoffDelay = BACKOFF_BASE;
        if (pendingPayload) {
            ws.send(JSON.stringify(pendingPayload));
            pendingPayload = null;
            ready = false;
            showSpinner();
        }
        updateConnectionStatus(true);
    };
    ws.onclose = () => {
        updateConnectionStatus(false);
        addSystemMessage(`Disconnected from ${uri}. Reconnecting...`);
        attempt++;
        setTimeout(connect, backoffDelay);
        backoffDelay = Math.min(backoffDelay * 2, BACKOFF_MAX);
        ready = true;
        hideSpinner();
    };
    ws.onmessage = (event) => handleMessage(JSON.parse(event.data));
}
```

3. **Message handler (dispatches to render functions — stubs for now):**
```javascript
function handleMessage(data) {

    if (data.content) handleContent(data.content);

}
```

```javascript
const input = document.querySelector('.chat-input input');
input.addEventListener('keydown', (e) => {
    if (e.key !== 'Enter' || !ready) return;
    const text = input.value.trim();
    if (!text) return;
    input.value = '';

    if (text === '/verbose') {
        verbose = !verbose;

    let payload;

        clearChat();
        clearArtifacts();
        clearWorkflow();

        payload = { content: text, command: null };

        return;
```

```javascript
function rebuildTree() {
    const tree = {};
    for (const [path] of artifacts) {
        const parts = path.split('/');
        let node = tree;
        for (let i = 0; i < parts.length - 1; i++) {
            if (!node[parts[i]]) node[parts[i]] = {};
            node = node[parts[i]];
        }
        node[parts[parts.length - 1]] = path; // leaf = full path string
    }
```

```javascript
let currentWorkflow = null;

function handleWorkflow(workflow) {
    currentWorkflow = workflow;
    const bar = document.querySelector('.workflow-bar');
    bar.style.display = 'flex';
    bar.innerHTML = '<span class="dot" style="color:#4ade80;">●</span>';

    const currentIdx = workflow.steps.indexOf(workflow.current);
    workflow.steps.forEach((step, i) => {
        if (i > 0) {
            const arrow = document.createElement('span');
            arrow.className = 'arrow';
            arrow.textContent = '→';
            bar.appendChild(arrow);
        }
        const span = document.createElement('span');
        span.className = 'step';
        if (i < currentIdx) span.classList.add('completed');
        if (i === currentIdx) span.classList.add('active');
        span.textContent = step;
        bar.appendChild(span);
    });
}

function clearWorkflow() {
    currentWorkflow = null;
    const bar = document.querySelector('.workflow-bar');
    bar.style.display = 'none';
    bar.innerHTML = '';
}
```

```javascript
const resizeHandle = document.querySelector('.resize-handle');
const chatPane = document.querySelector('.chat-pane');
const artifactPane = document.querySelector('.artifact-pane');
let isResizing = false;

resizeHandle.addEventListener('mousedown', (e) => {
    isResizing = true;
    document.body.style.cursor = 'col-resize';
    document.body.style.userSelect = 'none';
    e.preventDefault();
});

document.addEventListener('mousemove', (e) => {
    if (!isResizing) return;
    const container = document.querySelector('.main');
    const containerRect = container.getBoundingClientRect();
    const minWidth = 300;
    let newLeft = e.clientX - containerRect.left;
    newLeft = Math.max(minWidth, Math.min(newLeft, containerRect.width - minWidth));
    const leftPct = (newLeft / containerRect.width) * 100;
    chatPane.style.flex = 'none';
    chatPane.style.width = `${leftPct}%`;
    artifactPane.style.flex = '1';
});

document.addEventListener('mouseup', () => {
    if (isResizing) {
        isResizing = false;
        document.body.style.cursor = '';
        document.body.style.userSelect = '';
    }
});
```
